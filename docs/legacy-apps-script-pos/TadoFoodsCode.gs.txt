/********************************
 * UTIL
 ********************************/
function headerMap(h) {
  const m = {};
  h.forEach((x,i)=>m[x.toString().trim().toLowerCase().replace(/\s+/g,'')]=i);
  return m;
}

function _sf(val) {
  const s = String(val == null ? '' : val).trim();
  return /^[=+\-@|%`]/.test(s) ? "'" + s : s;
}

function stockStatus(stock, reorder) {
  if (stock <= 0) return "Out of Stock";
  if (stock <= reorder) return "Need to Reorder";
  return "Available";
}

/* Returns the inventory sheet for a given store name.
 * Each store has its own sheet (e.g. "Wholesale", "Retail").
 * Throws a clear error if the sheet doesn't exist. */
function getStoreSheet(ss, storeName) {
  const sh = ss.getSheetByName(storeName);
  if (!sh) throw new Error('Store sheet not found: "' + storeName + '". Check that this store has its own sheet.');
  return sh;
}

/********************************
 * SALE ID (SHORT, CLEAN)
 ********************************/
function generateSaleID() {
  const d = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyMMdd");
  const unique = Utilities.getUuid().slice(0, 4).toUpperCase();
  return "HJ-" + d + "-" + unique;
}

/********************************
 * LOGIN
 * Returns userInfo + sessionToken on success, null on failure.
 * No auth check needed here — this IS the auth step.
 ********************************/
function checkCode(o){
  // o = { code, nonce }
  const code  = String(o.code  || '').trim();
  const nonce = String(o.nonce || '');

  // ── Brute-force lockout check ──────────────────────────────────
  const bfStatus = _bfCheck(nonce);
  if (bfStatus.locked) {
    return { locked: true, msg: 'Too many failed attempts — please wait 15 minutes before trying again' };
  }

  const sh = SpreadsheetApp.getActive().getSheetByName("SalesPerson");
  const d = sh.getDataRange().getValues();
  const h = headerMap(d[0]);

  const roleDefaults = {
    'staff':    ['sales','returns'],
    'manager':  ['sales','returns','approve_returns','expenses','stock','prices','dashboard_ops','activity','barcode','attendance','customers'],
    'admin':    ['sales','returns','approve_returns','expenses','stock','prices','dashboard_ops','dashboard_finance','activity','barcode','attendance','staff_mgmt','settings','customers']
  };

  for(let i = 1; i < d.length; i++){
    {
      const stored = String(d[i][h.logincode]).trim();
      const match  = _isHashed(stored)
        ? stored === _hashPin(code)   // hashed comparison
        : stored === code;            // legacy plaintext (lazy migration)
      if(match && d[i][h.status] === 'Active'){
        // Silently upgrade plaintext PIN to hash on first successful login
        if (!_isHashed(stored)) {
          sh.getRange(i + 1, h.logincode + 1).setValue(_hashPin(code));
        }
      const role = (d[i][h.role] || 'Staff').toLowerCase();
      const isAdmin   = role === 'admin';
      const isManager = role === 'manager';

      let permissions = roleDefaults[role] || roleDefaults['staff'];
      if(h.permissions !== undefined && d[i][h.permissions]){
        const custom = String(d[i][h.permissions]).split(',').map(p=>p.trim()).filter(Boolean);
        if(custom.length) permissions = custom;
      }
      if(h.extraperms !== undefined && d[i][h.extraperms]){
        const extra = String(d[i][h.extraperms]).split(',').map(p=>p.trim()).filter(Boolean);
        extra.forEach(p=>{ if(!permissions.includes(p)) permissions.push(p); });
      }

      const userInfo = {
        name:            d[i][h.name],
        store:           d[i][h.store],
        role:            role,
        isAdmin:         isAdmin,
        isManager:       isManager,
        permissions:     permissions,
        availableStores: (isAdmin || isManager) ? getAllStores() : [d[i][h.store]]
      };

      // ── NEW: create server-side session and return its token ──
      const sessionToken = _createSession(userInfo);
      _bfReset(nonce); // clear failed-attempt counter on success
      return { ...userInfo, sessionToken };
      } // closes match && Active
    }   // closes row loop block
  }
  // No matching PIN found — record failed attempt
  const bf = _bfFail(nonce);
  _bfThrottle();   // #26: global soft-throttle that a rotating nonce can't dodge
  if (bf.locked) {
    return { locked: true, msg: 'Too many failed attempts — please wait 15 minutes before trying again' };
  }
  return null;
}

/********************************
 * GET ALL STORES (INTERNAL — called from checkCode, not exposed directly)
 ********************************/
function getAllStores(){
  const sh = SpreadsheetApp.getActive().getSheetByName("SalesPerson");
  const d = sh.getDataRange().getValues();
  const h = headerMap(d[0]);
  const stores = [];
  for(let i = 1; i < d.length; i++){
    const store = d[i][h.store];
    if(store && !stores.includes(store)) stores.push(store);
  }
  return stores.sort();
}

/********************************
 * LOAD PRODUCTS
 * Signature change: getProducts({store, _token})
 ********************************/
function getProducts(o){
  const sess = _verifySession(o._token, 'sales');
  if(!sess) return _unauth();
  const store = o.store;

  const sh = getStoreSheet(SpreadsheetApp.getActive(), store);
  const d = sh.getDataRange().getValues();
  const h = headerMap(d[0]);

  return d.slice(1)
    .filter(r => r[h.productname])
    .map(r => ({
      id:           r[h.productid],
      name:         r[h.productname],
      category:     r[h.category],
      subCategory:  r[h.subcategory],
      price:        Number(r[h.listprice]),
      cost:         Number(r[h.cost]) || 0,
      expenseOnCost:Number(r[h["expenseoncost"]]) || 0,
      stock:        Number(r[h.stock]),
      reorder:      Number(r[h.reorderlevel]),
      expiryDate:   h.expirydate!==undefined && r[h.expirydate] ? Utilities.formatDate(new Date(r[h.expirydate]),Session.getScriptTimeZone(),'yyyy-MM-dd') : '',
      daysToExpiry: h.expirydate!==undefined && r[h.expirydate] ? Math.round((new Date(r[h.expirydate])-new Date())/86400000) : null
    }));
}

/********************************
 * RECORD SALE (INVENTORY = OUT)
 * o must include _token
 ********************************/
function recordSale(o){
  const sess = _verifySession(o._token, 'sales');
  if(!sess) return _unauth();

  const ss = SpreadsheetApp.getActive();

  // Serialize the read-check-write so two cashiers can't oversell the last unit (#21).
  const lock = LockService.getScriptLock();
  try { lock.waitLock(8000); }
  catch(e){ return { msg: 'System busy recording another sale — please try again' }; }
  try {

  const prod  = getStoreSheet(ss, o.store);
  const sales = ss.getSheetByName("FactSales");
  const inv   = ss.getSheetByName("FactInventory");

  const d = prod.getDataRange().getValues();
  const h = headerMap(d[0]);

  let r = -1;
  for(let i = 1; i < d.length; i++){
    if (d[i][h.productid] === o.productID) {
      r = i;
      break;
    }
  }
  if (r === -1) return { msg: "Product not found" };

  const qty   = Number(o.qty);
  const price = Number(o.unitSoldPrice);
  if (!Number.isFinite(qty)   || qty   < 1 || qty   > 10000) return { msg: 'Invalid quantity' };
  if (!Number.isFinite(price) || price < 0 || price > 1e9)   return { msg: 'Invalid unit price' };
  const stock = Number(d[r][h.stock]);
  if (stock < qty) return { msg: "Insufficient stock" };

  // #18: fold this line's share of the cart discount into the stored amount so revenue is net.
  const disc      = Math.max(0, Number(o.disc) || 0);
  const lineTotal = Math.max(0, price * qty - disc);

  const newStock = stock - qty;
  const status   = stockStatus(newStock, Number(d[r][h.reorderlevel]));
  const saleID   = o.saleID || generateSaleID();
  const txDate   = o.date ? new Date(o.date + 'T' + new Date().toTimeString().slice(0,8)) : new Date();
  const totalCost = (Number(d[r][h.cost]) + Number(d[r][h["expenseoncost"]])) * qty;

  sales.appendRow([
    saleID, txDate, o.productID, qty, d[r][h.listprice],
    totalCost, price, lineTotal,
    _sf(o.payment), _sf(o.soldBy), _sf(o.store), o.vat||0, o.vatRate||0
  ]);

  if(o.customerName && o.customerName.trim()){
    _upsertCustomer(_sf(o.customerName.trim()), _sf(o.customerWA||''), lineTotal,
      d[r][h.productname]||o.productID, o.store, saleID, o.date);
  }

  prod.getRange(r + 1, h.stock + 1).setValue(newStock);
  prod.getRange(r + 1, h.status + 1).setValue(status);

  checkStockAlert(o.productID, d[r][h.productname]||o.productID, newStock,
    Number(d[r][h.reorderlevel]), o.store);

  inv.appendRow([
    "INV-" + saleID, txDate, o.productID, "OUT", qty, newStock, o.soldBy, o.store
  ]);

  return { msg: "Sale completed successfully", saleID: saleID };

  } finally { lock.releaseLock(); }
}

/********************************
 * WINDOW BLIND SALE LOGIC (internal — called from recordSale, already authed)
 ********************************/
function recordWindowBlindSale(o){
  const ss  = SpreadsheetApp.getActive();
  const dim   = ss.getSheetByName("DimWindowBlind");
  const sales = ss.getSheetByName("FactSales");
  const inv   = ss.getSheetByName("FactInventory");

  const data = dim.getDataRange().getValues();
  const h    = headerMap(data[0]);

  const saleID = o.saleID || generateSaleID();
  const txDate = o.date ? new Date(o.date + 'T' + new Date().toTimeString().slice(0,8)) : new Date();
  const sqmUsed = Number(o.width) * Number(o.height) * Number(o.qty || 1);

  let fabricRow = -1;
  for (let i = 1; i < data.length; i++) {
    if (data[i][h.productid] === o.fabricID && data[i][h.store] === o.store) {
      fabricRow = i;
      break;
    }
  }
  if (fabricRow === -1) return { msg: "Fabric not found" };

  const currentFabricStock = Number(data[fabricRow][h.stock]);
  if (currentFabricStock < sqmUsed) return { msg: "Not enough fabric stock" };
  const newFabricStock = currentFabricStock - sqmUsed;

  dim.getRange(fabricRow + 1, h.stock + 1).setValue(newFabricStock);
  dim.getRange(fabricRow + 1, h.status + 1)
     .setValue(stockStatus(newFabricStock, Number(data[fabricRow][h.reorderlevel])));

  inv.appendRow([
    "INV-" + saleID, txDate, o.fabricID, "OUT", sqmUsed, newFabricStock, o.soldBy, o.store
  ]);

  for (let item of o.accessories) {
    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (
        String(data[i][h.productname]).trim().toLowerCase() ===
        String(item.name).trim().toLowerCase() &&
        data[i][h.store] === o.store
      ) { rowIndex = i; break; }
    }
    if (rowIndex === -1) return { msg: "Accessory not found: " + item.name };

    const accessoryProductID = data[rowIndex][h.productid];
    const currentStock       = Number(data[rowIndex][h.stock]);
    if (currentStock < item.qty) return { msg: "Not enough stock for " + item.name };

    const updatedStock = currentStock - item.qty;
    dim.getRange(rowIndex + 1, h.stock + 1).setValue(updatedStock);
    dim.getRange(rowIndex + 1, h.status + 1)
       .setValue(stockStatus(updatedStock, Number(data[rowIndex][h.reorderlevel])));

    inv.appendRow([
      "INV-" + saleID + "-" + accessoryProductID, txDate,
      accessoryProductID, "OUT", item.qty, updatedStock, o.soldBy, o.store
    ]);
  }

  let totalCost = 0;
  const dimForCost = dim.getDataRange().getValues();
  const hc = headerMap(dimForCost[0]);

  const costItems = [
    { name: data[fabricRow][hc.productname], qty: sqmUsed },
    { name: "Rail",        qty: Number(o.width) * Number(o.qty || 1) },
    { name: "Miler",       qty: Number(o.width) * Number(o.qty || 1) },
    { name: "Small Belt",  qty: Number(o.width) * Number(o.qty || 1) },
    { name: "Big Belt",    qty: Number(o.width) * Number(o.qty || 1) },
    { name: "Accessories", qty: Number(o.accessories.find(a => a.name === "Accessories")?.qty || 1) }
  ];
  for(const item of costItems){
    const found = dimForCost.slice(1).find(r =>
      String(r[hc.productname]).trim().toLowerCase() === item.name.trim().toLowerCase() &&
      String(r[hc.store]).trim() === o.store
    );
    if(found) totalCost += (Number(found[hc.cost]) || 0) * item.qty;
  }

  sales.appendRow([
    saleID, txDate, o.fabricID, sqmUsed, 0, totalCost, o.finalPrice, o.finalPrice,
    o.payment, o.soldBy, o.store, o.vat||0, o.vatRate||0
  ]);

  if(o.customerName && o.customerName.trim()){
    _upsertCustomer(o.customerName.trim(), o.customerWA||'', o.finalPrice,
      data[fabricRow][h.productname]||o.fabricID, o.store, saleID, o.date);
  }

  return { msg: "Window blind sale recorded successfully", saleID: saleID };
}

/********************************
 * SELL NEW PRODUCT
 * Requires 'settings' permission
 ********************************/
function sellNewProduct(o){
  const sess = _verifySession(o._token, 'settings');
  if(!sess) return _unauth();

  const ss   = SpreadsheetApp.getActive();
  const inv  = ss.getSheetByName("FactInventory");

  if(
    !o.productName?.toString().trim() ||
    !o.category?.toString().trim()    ||
    !o.subCategory?.toString().trim() ||
    !o.store?.toString().trim()
  ) return { msg: "Missing required product information" };

  const initialStock = Number(o.initialStock);
  const qty          = Number(o.qty);
  const unitPrice    = Number(o.unitSoldPrice);

  if(!Number.isFinite(initialStock) || initialStock < 0 || initialStock > 100000)
    return { msg: "Initial stock must be between 0 and 100,000" };
  if(!Number.isFinite(qty) || qty < 0)
    return { msg: "Quantity cannot be negative" };
  if(!Number.isFinite(unitPrice) || unitPrice <= 0 || unitPrice > 1e9)
    return { msg: "Selling price must be between ₦1 and ₦1,000,000,000" };
  if(qty > initialStock)
    return { msg: "Cannot sell more than available stock" };

  const prod = getStoreSheet(ss, o.store);
  const existingData  = prod.getDataRange().getValues();
  const existingH     = headerMap(existingData[0]);
  const existingNames = existingData.slice(1).map(r=>String(r[existingH.productname]||'').toLowerCase().trim());
  if(existingNames.includes(o.productName.trim().toLowerCase()))
    return { msg: "Product already exists with this name" };

  const productID      = "P-" + Utilities.getUuid().slice(0,6).toUpperCase();
  const remainingStock = initialStock - qty;
  const reorderLevel   = Number(o.reorderLevel) || 1;
  const status         = stockStatus(remainingStock, reorderLevel);
  const txDate         = o.date ? new Date(o.date + 'T' + new Date().toTimeString().slice(0,8)) : new Date();

  // Write product to the chosen store sheet
  prod.appendRow([
    productID, _sf(o.productName.trim()), _sf(o.category.trim()), _sf(o.subCategory.trim()),
    _sf(o.store), Number(o.cost)||0, Number(o.expenseOnCost)||0, unitPrice,
    remainingStock, reorderLevel, status
  ]);
  // Write expiry date and alert days if provided
  if(o.expiryDate || o.alertDays){
    const newRow = prod.getLastRow();
    if(o.expiryDate && existingH.expirydate !== undefined){
      prod.getRange(newRow, existingH.expirydate+1).setValue(o.expiryDate);
    }
    if(o.alertDays && existingH.expiryalertdays !== undefined){
      prod.getRange(newRow, existingH.expiryalertdays+1).setValue(o.alertDays);
    }
  }

  if(initialStock > 0){
    inv.appendRow([
      "INV-INITIAL-" + productID, txDate, productID, "IN",
      initialStock, initialStock, o.soldBy, o.store
    ]);
  }

  // Auto-create matching 0-stock row in every OTHER store (so transfers work)
  const allStores = getAllStores();
  allStores.forEach(otherStore => {
    if (otherStore === o.store) return; // skip current store
    const otherSheet = ss.getSheetByName(otherStore);
    if (!otherSheet) return; // skip if that store sheet doesn't exist
    const otherData = otherSheet.getDataRange().getValues();
    const otherH = headerMap(otherData[0]);
    const otherNames = otherData.slice(1).map(r=>String(r[otherH.productname]||'').toLowerCase().trim());
    if (otherNames.includes(o.productName.trim().toLowerCase())) return; // skip if already exists
    otherSheet.appendRow([
      productID, _sf(o.productName.trim()), _sf(o.category.trim()), _sf(o.subCategory.trim()),
      _sf(otherStore), Number(o.cost)||0, Number(o.expenseOnCost)||0, unitPrice,
      0, reorderLevel, 'Out of Stock'
    ]);
  });

  return { msg: "Product created successfully" };
}

/********************************
 * UNDO SALE (RESTORE STOCK + FIX INVENTORY)
 * Signature change: undoLastSale({store, user, _token})
 * Also accepts saleID for precision undos
 ********************************/
function undoLastSale(o){
  const sess = _verifySession(o._token, 'sales');
  if(!sess) return _unauth();

  const store = o.store;
  const user  = o.user;

  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(8000); // wait up to 8 s; throws if another undo is in progress
  } catch(e) {
    return { msg: 'Another undo is in progress — please wait a moment and try again' };
  }
  try {

  const ss    = SpreadsheetApp.getActive();
  const sales   = ss.getSheetByName("FactSales");
  const archive = ss.getSheetByName("ArchiveSales");
  const prod    = getStoreSheet(ss, store);
  const inv     = ss.getSheetByName("FactInventory");

  const salesData = sales.getDataRange().getValues();
  if (salesData.length <= 1) return { msg: "No sale to undo" };

  const isMine = (row) =>
    String(row[9]).trim()  === String(user).trim() &&
    String(row[10]).trim() === String(store).trim();

  // ── Decide which line(s) to undo ──────────────────────────────────
  // #22: when a txnBase (the shared transaction id) is supplied, undo EVERY line of
  // that transaction; otherwise undo a single row (explicit saleID, else most recent).
  const undoRows = [];
  if (o.txnBase && !o.saleID) {
    const base = String(o.txnBase).trim();
    for (let i = 1; i < salesData.length; i++) {
      const id = String(salesData[i][0]).trim();
      if ((id === base || id.startsWith(base + '-')) && isMine(salesData[i])) {
        undoRows.push({ rowIndex: i, sale: salesData[i] });
      }
    }
  }
  if (!undoRows.length) {
    let anchor = -1;
    if (o.saleID) {
      for (let i = salesData.length - 1; i > 0; i--) {
        if (String(salesData[i][0]).trim() === String(o.saleID).trim()) { anchor = i; break; }
      }
      if (anchor !== -1) {
        if (String(salesData[anchor][9]).trim()  !== String(user).trim())
          return { msg: "You can only undo your own sales" };
        if (String(salesData[anchor][10]).trim() !== String(store).trim())
          return { msg: "Sale belongs to a different store" };
      }
    } else {
      for (let i = salesData.length - 1; i > 0; i--) {
        if (isMine(salesData[i])) { anchor = i; break; }
      }
    }
    if (anchor === -1) return { msg: "No sale found for you to undo" };
    undoRows.push({ rowIndex: anchor, sale: salesData[anchor] });
  }

  // ── Restore stock + archive each line ─────────────────────────────
  const prodData = prod.getDataRange().getValues();
  const h = headerMap(prodData[0]);
  const touched = {};

  undoRows.forEach(u => {
    const sale = u.sale;
    const productID = sale[2];
    const qty = Number(sale[3]) || 0;
    for (let i = 1; i < prodData.length; i++) {
      if (prodData[i][h.productid] === productID) {
        const restored = Number(prodData[i][h.stock]) + qty;
        prodData[i][h.stock] = restored;                 // keep local copy in sync across lines
        prod.getRange(i + 1, h.stock + 1).setValue(restored);
        prod.getRange(i + 1, h.status + 1)
            .setValue(stockStatus(restored, Number(prodData[i][h.reorderlevel])));
        break;
      }
    }
    touched[productID] = true;
    archive.appendRow([
      sale[0], sale[1], sale[2], sale[3], sale[4],
      sale[6], sale[7], sale[8], sale[9], sale[10],
      new Date(), user
    ]);
  });

  // ── Delete matching inventory OUT rows (collect, then delete bottom-up) ──
  const invData = inv.getDataRange().getValues();
  const invToDelete = [];
  undoRows.forEach(u => {
    const sID = String(u.sale[0]).trim();
    for (let i = invData.length - 1; i > 0; i--) {
      if (String(invData[i][0]) === "INV-" + sID &&
          String(invData[i][3]) === "OUT" &&
          invData[i][7] === store && invData[i][2] &&
          invToDelete.indexOf(i) === -1) {
        invToDelete.push(i); break;
      }
    }
  });
  if (invToDelete.length) {
    invToDelete.sort((a,b)=>b-a).forEach(i => inv.deleteRow(i + 1));
    SpreadsheetApp.flush();
    Object.keys(touched).forEach(pid => recalculateInventoryBalances(pid, store));
  }

  // ── Delete the sale rows (bottom-up so row numbers stay valid) ─────
  undoRows.map(u => u.rowIndex).sort((a,b)=>b-a).forEach(i => sales.deleteRow(i + 1));

  const n = undoRows.length;
  return { msg: n > 1 ? ('Last sale undone — ' + n + ' items · stock restored') : 'Last sale undone and stock restored' };
  } finally { lock.releaseLock(); }
}

/********************************
 * UNDO WINDOW BLIND SALE (internal — already authed)
 ********************************/
function undoWindowBlindSale(o){
  const store = o.store;
  const user  = o.user;

  const lock = LockService.getScriptLock();
  try { lock.waitLock(8000); } catch(e) {
    return { msg: 'Another undo is in progress — please try again shortly' };
  }
  try {

  const ss      = SpreadsheetApp.getActive();
  const sales   = ss.getSheetByName("FactSales");
  const archive = ss.getSheetByName("ArchiveSales");
  const dim     = ss.getSheetByName("DimWindowBlind");
  const inv     = ss.getSheetByName("FactInventory");

  const salesData = sales.getDataRange().getValues();
  let saleRow = -1, sale;
  for (let i = salesData.length - 1; i > 0; i--) {
    if (salesData[i][9] === store && salesData[i][8] === user) {
      saleRow = i; sale = salesData[i]; break;
    }
  }
  if (saleRow === -1) return { msg: "No window blind sale found to undo" };

  const saleID       = sale[0];
  const invData      = inv.getDataRange().getValues();
  const qtyToRestore = {};
  const rowsToDelete = [];

  for (let i = 1; i < invData.length; i++) {
    const invID    = String(invData[i][0]);
    const prodID   = invData[i][2];
    const type     = invData[i][3];
    const invStore = invData[i][7];
    const belongs  = invID === "INV-" + saleID || invID.startsWith("INV-" + saleID + "-WB-");
    if (belongs && type === "OUT" && invStore === store) {
      rowsToDelete.push(i + 1);
      qtyToRestore[prodID] = (qtyToRestore[prodID] || 0) + Number(invData[i][4]);
    }
  }

  rowsToDelete.sort((a,b)=>b-a).forEach(r=>inv.deleteRow(r));
  SpreadsheetApp.flush();

  const dimData = dim.getDataRange().getValues();
  const h = headerMap(dimData[0]);
  for (const [prodID, deletedQty] of Object.entries(qtyToRestore)) {
    for (let i = 1; i < dimData.length; i++) {
      if (dimData[i][h.productid] === prodID && dimData[i][h.store] === store) {
        const newStock = Number(dimData[i][h.stock]) + deletedQty;
        dim.getRange(i+1, h.stock+1).setValue(newStock);
        dim.getRange(i+1, h.status+1)
           .setValue(stockStatus(newStock, Number(dimData[i][h.reorderlevel])));
        break;
      }
    }
  }

  SpreadsheetApp.flush();
  for (const prodID of Object.keys(qtyToRestore)) {
    recalculateWindowBlindInventory(prodID, store, dim);
  }

  archive.appendRow([
    sale[0], sale[1], sale[2], sale[3], sale[4],
    sale[6], sale[7], sale[8], sale[9], sale[10],
    new Date(), user
  ]);
  sales.deleteRow(saleRow + 1);

}

function recalculateWindowBlindInventory(productID, store, dim) {
  const ss  = SpreadsheetApp.getActive();
  const inv = ss.getSheetByName("FactInventory");
  const invData = inv.getDataRange().getValues();
  const dimData = dim.getDataRange().getValues();
  const h = headerMap(dimData[0]);

  let currentDimStock = 0;
  for (let i = 1; i < dimData.length; i++) {
    if (dimData[i][h.productid] === productID && dimData[i][h.store] === store) {
      currentDimStock = Number(dimData[i][h.stock]); break;
    }
  }

  const outRows = [];
  for (let i = 1; i < invData.length; i++) {
    if (invData[i][2] === productID && invData[i][7] === store && invData[i][3] === "OUT") {
      outRows.push({ sheetRow: i+1, date: new Date(invData[i][1]), qty: Number(invData[i][4]) });
    }
  }
  if (outRows.length === 0) return;

  outRows.sort((a,b)=>a.date-b.date);
  const totalOuts = outRows.reduce((sum,r)=>sum+r.qty, 0);
  let runningBalance = currentDimStock + totalOuts;
  for (const row of outRows) {
    runningBalance -= row.qty;
    inv.getRange(row.sheetRow, 6).setValue(runningBalance);
  }
}

function recalculateInventoryBalances(productID, store) {
  const ss  = SpreadsheetApp.getActive();
  const inv = ss.getSheetByName("FactInventory");
  let dim;
  try { dim = getStoreSheet(ss, store); }        // #23: read the correct store sheet, not DimProduct
  catch(e){ dim = ss.getSheetByName("DimProduct"); }
  if(!dim) return;
  const invData = inv.getDataRange().getValues();
  const dimData = dim.getDataRange().getValues();
  const h = headerMap(dimData[0]);

  let currentDimStock = 0;
  for (let i = 1; i < dimData.length; i++) {
    if (dimData[i][h.productid] === productID && dimData[i][h.store] === store) {
      currentDimStock = Number(dimData[i][h.stock]); break;
    }
  }

  const rows = [];
  for (let i = 1; i < invData.length; i++) {
    if (invData[i][2] === productID && invData[i][7] === store) {
      rows.push({ sheetRow: i+1, date: new Date(invData[i][1]), type: invData[i][3], qty: Number(invData[i][4]) });
    }
  }
  if (rows.length === 0) return;

  rows.sort((a,b)=>a.date-b.date);
  const totalOuts = rows.filter(r=>r.type==="OUT").reduce((s,r)=>s+r.qty, 0);
  const totalIns  = rows.filter(r=>r.type==="IN").reduce((s,r)=>s+r.qty, 0);
  let runningBalance = currentDimStock + totalOuts - totalIns;

  for (const row of rows) {
    runningBalance += row.type === "IN" ? row.qty : -row.qty;
    inv.getRange(row.sheetRow, 6).setValue(runningBalance);
  }
}

/********************************
 * STOCK IN (INVENTORY = IN)
 * o must include _token
 ********************************/
function stockIn(o){
  const sess = _verifySession(o._token, 'stock');
  if(!sess) return _unauth();
  const qty = Number(o.qty);
  if(!Number.isInteger(qty) || qty < 1 || qty > 100000)
    return { msg: 'Invalid quantity — must be a whole number between 1 and 100,000' };

  const ss   = SpreadsheetApp.getActive();
  const prod = getStoreSheet(ss, o.store);
  const inv  = ss.getSheetByName("FactInventory");
  const d    = prod.getDataRange().getValues();
  const h    = headerMap(d[0]);
  const txDate = o.date ? new Date(o.date + 'T' + new Date().toTimeString().slice(0,8)) : new Date();

  for(let i = 1; i < d.length; i++){
    if (d[i][h.productid] === o.productID) {
      const newStock = Number(d[i][h.stock]) + o.qty;
      prod.getRange(i+1, h.stock+1).setValue(newStock);
      prod.getRange(i+1, h.status+1).setValue(stockStatus(newStock, Number(d[i][h.reorderlevel])));
      inv.appendRow([
        "INV-IN-" + Utilities.getUuid().slice(0,6), txDate,
        o.productID, "IN", o.qty, newStock, o.performedBy, o.store
      ]);
      // Handle expiry date
      var oldExpiryNote = null;
      if(o.expiryDate && h.expirydate !== undefined){
        var oldExpRaw = d[i][h.expirydate];
        var newExp = new Date(o.expiryDate);
        if(oldExpRaw){
          var oldExp = new Date(oldExpRaw);
          oldExp.setHours(0,0,0,0);
          var todayX = new Date(); todayX.setHours(0,0,0,0);
          var oldDays = Math.round((oldExp - todayX) / 86400000);
          if(newExp > oldExp && oldDays <= 60){
            oldExpiryNote = {
              productName: String(d[i][h.productname] || o.productID),
              oldExpiry: Utilities.formatDate(oldExp, Session.getScriptTimeZone(), 'yyyy-MM-dd'),
              daysToExpiry: oldDays,
              store: o.store
            };
          }
        }
        if(!oldExpRaw || newExp > new Date(oldExpRaw)){
          prod.getRange(i+1, h.expirydate+1).setValue(o.expiryDate);
        }
      }
      if(o.alertDays && h.expiryalertdays !== undefined){
        prod.getRange(i+1, h.expiryalertdays+1).setValue(o.alertDays);
      }
      return { msg: "Stock added successfully", oldExpiryNote: oldExpiryNote };
    }
  }
  return { msg: "Product not found" };
}

/********************************
 * TRANSFER STOCK BETWEEN STORES
 * Requires admin or manager role
 ********************************/
function transferStock(o){
  const sess = _verifySession(o._token);
  if(!sess) return _unauth();
  if(sess.role !== 'admin' && sess.role !== 'manager') return _unauth();

  const fromQty = Number(o.fromQty);
  const toQty   = Number(o.toQty);
  if(!fromQty || fromQty < 1) return { msg: 'Invalid source quantity' };
  if(!toQty   || toQty   < 1) return { msg: 'Invalid destination quantity' };
  if(o.fromStore === o.toStore) return { msg: 'Source and destination stores must be different' };

  const ss  = SpreadsheetApp.getActive();
  const inv = ss.getSheetByName('FactInventory');
  const now = new Date();

  // ── DEDUCT from source ──────────────────────────────────────
  const fromSheet = getStoreSheet(ss, o.fromStore);
  const fromData  = fromSheet.getDataRange().getValues();
  const fh = headerMap(fromData[0]);
  let fromRow = -1;
  for(let i = 1; i < fromData.length; i++){
    if(String(fromData[i][fh.productid]).trim() === String(o.fromProductID).trim()){
      fromRow = i; break;
    }
  }
  if(fromRow === -1) return { msg: 'Source product not found in ' + o.fromStore };
  const fromStock = Number(fromData[fromRow][fh.stock]);
  if(fromStock < fromQty) return { msg: 'Insufficient stock in ' + o.fromStore + ' — only ' + fromStock + ' available' };

  const newFromStock = fromStock - fromQty;
  fromSheet.getRange(fromRow + 1, fh.stock + 1).setValue(newFromStock);
  fromSheet.getRange(fromRow + 1, fh.status + 1).setValue(stockStatus(newFromStock, Number(fromData[fromRow][fh.reorderlevel])));

  // ── ADD to destination ────────────────────────────────────
  const toSheet = getStoreSheet(ss, o.toStore);
  const toData  = toSheet.getDataRange().getValues();
  const th = headerMap(toData[0]);
  let toRow = -1;
  for(let i = 1; i < toData.length; i++){
    if(String(toData[i][th.productid]).trim() === String(o.toProductID).trim()){
      toRow = i; break;
    }
  }
  if(toRow === -1) return { msg: 'Destination product not found in ' + o.toStore };

  const toStock    = Number(toData[toRow][th.stock]);
  const newToStock = toStock + toQty;
  toSheet.getRange(toRow + 1, th.stock + 1).setValue(newToStock);
  toSheet.getRange(toRow + 1, th.status + 1).setValue(stockStatus(newToStock, Number(toData[toRow][th.reorderlevel])));

  // ── Log both movements in FactInventory ───────────────────
  const txID = 'TRF-' + Utilities.getUuid().slice(0,6).toUpperCase();
  inv.appendRow([ 'INV-' + txID + '-OUT', now, o.fromProductID, 'TRANSFER_OUT', fromQty, newFromStock, o.performedBy, o.fromStore ]);
  inv.appendRow([ 'INV-' + txID + '-IN',  now, o.toProductID,  'TRANSFER_IN',  toQty,  newToStock,  o.performedBy, o.toStore  ]);

  return {
    msg: 'Transfer complete · ' + fromQty + ' out of ' + o.fromStore + ' · ' + toQty + ' into ' + o.toStore
  };
}

/********************************

  let filterStore = o.store || null;

  const normalizeStore = s => String(s||'').trim().toLowerCase().replace(/\s+/g,'');
  const inRange = date => {
    if(!date) return false;
    const d = (date instanceof Date) ? date : new Date(date);
    if(isNaN(d.getTime())) return false;
    if(fromDate && d < fromDate) return false;
    if(toDate   && d > toDate)   return false;

    storeMap[store].sales++; storeMap[store].revenue+=revenue; storeMap[store].profit+=profit;

    if(!sellerMap[soldBy]) sellerMap[soldBy]={name:soldBy,revenue:0,sales:0};
    sellerMap[soldBy].revenue+=revenue; sellerMap[soldBy].sales++;

    if(payMap[pay]!==undefined) payMap[pay]+=revenue; else payMap[pay]=revenue;

    if(!monthlyMap[monthKey]) monthlyMap[monthKey]={month:monthKey,revenue:0,transactions:0};
    monthlyMap[monthKey].revenue+=revenue; monthlyMap[monthKey].transactions++;
  });

  const expBreak={};
  expData.slice(1).forEach(r=>{
    const rawDate = r[eh.date||1];
    const date    = (rawDate instanceof Date) ? rawDate : new Date(rawDate);
    const store   = String(r[eh.store||6]||'');
    if(filterStore && normalizeStore(store) !== normalizeStore(filterStore)) return;
    const amt     = Number(r[eh.amount||3])||0;
    const dateStr = Utilities.formatDate(date, tz, "yyyy-MM-dd");
    if(dateStr===todayStr) todayExpenses+=amt;
    if(storeMap[store]) storeMap[store].expenses=(storeMap[store].expenses||0)+amt;
    const type = String(r[eh.expensetype||2]||'Other');
    if(inRange(date)) expBreak[type]=(expBreak[type]||0)+amt;
  });

  const lowStock = [];
  allProdRows.forEach(item=>{
    const r = item.row;
    const ph = item.h;
    if(filterStore && item.store!==filterStore) return;
    if(Number(r[ph.stock])<=Number(r[ph.reorderlevel]))
      lowStock.push({name:r[ph.productname],stock:Number(r[ph.stock]),store:item.store});
  });

  const days          = ['Mon','Tue','Wed','Thu','Fri','Sat','Sun'];
  const monthlyTrend  = Object.values(monthlyMap).sort((a,b)=>a.month.localeCompare(b.month)).slice(-12);
  const allProds      = Object.values(topMap).sort((a,b)=>b.revenue-a.revenue);
  const dailyTrend    = Object.values(dailyMap).sort((a,b)=>a.date.localeCompare(b.date));
  const avgSaleValue  = totalTransactions>0?Math.round(totalRevenue/totalTransactions):0;
  const totalExpenses = Object.values(expBreak).reduce((s,v)=>s+v,0);
  const hourlyData    = Array.from({length:24},(_,h)=>hourlyMap[h]||{hour:h,revenue:0,count:0});

  return {
    todaySales, todayRevenue, todayProfit, todayExpenses,
    hourlyData,
    totalRevenue, totalProfit, totalTransactions, avgSaleValue, totalExpenses,
    weeklySales:     days.map(d=>weekly[d]||{day:d,revenue:0,count:0}),
    topProducts:     allProds.slice(0,5),
    bottomProducts:  allProds.slice(-5).reverse(),
    allProducts:     allProds,
    dailyTrend,
    lowStock:        lowStock.sort((a,b)=>a.stock-b.stock).slice(0,10),
    expenseBreakdown:Object.entries(expBreak).map(([type,amount])=>({type,amount})).sort((a,b)=>b.amount-a.amount),
    storeComparison: Object.values(storeMap),
    topSellers:      Object.values(sellerMap).sort((a,b)=>b.revenue-a.revenue).slice(0,8),
    paymentMethods:  payMap,
    revTrend:        monthlyTrend,
    allStores:       getAllStores(),
    totalExpenses
  };
}

/********************************
 * SALESPERSON ACTIVITY
 * Staff may only query their own name/store
 ********************************/
function getSalespersonActivity(o){
  const sess = _verifySession(o._token, 'activity');
  if(!sess) return _unauth();

  // Prevent staff from querying other people's activity
  if(!sess.isAdmin && !sess.isManager) {
    if(o.soldBy !== sess.name)  o.soldBy  = sess.name;
    if(o.store  !== sess.store) o.store   = sess.store;
  }

  const ss    = SpreadsheetApp.getActive();
  const sales = ss.getSheetByName("FactSales");

  const data  = sales.getDataRange().getValues();
  const h     = headerMap(data[0]);

  // Build name map from all store sheets
  const nameMap = {};
  getAllStores().forEach(storeName=>{
    const sh = ss.getSheetByName(storeName);
    if(!sh) return;
    const sd = sh.getDataRange().getValues();
    if(sd.length < 2) return;
    const ph = headerMap(sd[0]);
    sd.slice(1).forEach(r=>{
      const id = String(r[ph.productid]||'').trim();
      if(id) nameMap[id] = String(r[ph.productname]||'').trim();
    });
  });
  
// (DimWindowBlind removed — name mapping uses DimProduct only)

  const cutoff = new Date();
  const days   = o.days || 14;
  cutoff.setDate(cutoff.getDate() - days);

  const rows = data.slice(1).filter(r=>{
    const rowUser  = String(r[9]||'').trim().toLowerCase();
    const rowStore = String(r[10]||'').trim().toLowerCase();
    const filterUser  = String(o.soldBy||'').trim().toLowerCase();
    const filterStore = String(o.store||'').trim().toLowerCase();
    if(rowUser!==filterUser || rowStore!==filterStore) return false;
    let d;
    if(r[1] instanceof Date) d=r[1];
    else d=new Date(r[1]);
    if(isNaN(d.getTime())) return false;
    if(o.month!==undefined&&o.month!==null&&o.year){
      return d.getMonth()===Number(o.month)&&d.getFullYear()===Number(o.year);
    }
    return d.getTime()>=cutoff.getTime();
  });

  const transactions = rows.map(r=>{
    const productID   = String(r[h.productid]||r[2]||'').trim();
    const productName = nameMap[productID] || productID;
    const isWB        = String(r[h.store]||r[10]||'').toLowerCase().includes('windowblind');
    const qty         = Number(r[h.quantity]||r[h.qty]||r[3])||1;
    const unitPrice   = Number(r[h.unitprice]||r[h.unitsoldprice]||r[6])||0;
    const total       = Number(r[h.totalamount]||r[7])||0;
    const rawDate     = r[h.date]||r[1];
    const dateStr     = rawDate instanceof Date
      ? Utilities.formatDate(rawDate, Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss")
      : String(rawDate||'');
    return {
      id:      String(r[h.saleid]||r[0]),
      date:    dateStr, time: dateStr,
      total, payment: String(r[h.paymenttype]||r[h.payment]||r[8]||'Cash'),
      soldBy:  String(r[h.soldby]||r[9]||''),
      store:   String(r[h.store]||r[10]||''),
      customer:'', discount:0,
      items:[{ name:productName, qty:isWB?parseFloat(qty.toFixed(2)):qty, unitPrice, isWB }]
    };
  });

  const grouped = {};
  transactions.forEach(t=>{
    if(!grouped[t.id]) grouped[t.id]={...t};
    else { grouped[t.id].items.push(...t.items); grouped[t.id].total+=t.total; }
  });

  return { transactions: Object.values(grouped) };
}

/** Restore returned units to the CORRECT store sheet (#23).
 *  Uses the store-named sheet like the rest of the app; falls back to DimProduct
 *  only for a pre-migration single-store layout. Columns resolved via headerMap. */
function _restoreReturnStock(ss, store, productID, qty){
  var sh;
  try { sh = getStoreSheet(ss, store); }
  catch(e){ sh = ss.getSheetByName('DimProduct'); }
  if(!sh) return false;
  var d = sh.getDataRange().getValues();
  var h = headerMap(d[0]);
  for(var i=1;i<d.length;i++){

      sh.getRange(i+1, h.stock+1).setValue(newStock);
      sh.getRange(i+1, h.status+1).setValue(stockStatus(newStock, Number(d[i][h.reorderlevel])||1));

  const sess = _verifySession(o._token, 'returns');
  if(!sess) return _unauth();

  const ss = SpreadsheetApp.getActive();

  if (!returnsSheet) {
    returnsSheet = ss.insertSheet('Returns');
    returnsSheet.appendRow([
      'ReturnID','Date','OriginalTxnID','ProductID','ProductName',
      'Qty','UnitPrice','RefundAmount','RefundMethod','Reason',
      'ReturnedBy','Store','Status','ApprovedBy','ApprovedDate'
    ]);
  }

  const returnId = 'RET-' + Utilities.getUuid().slice(0,6).toUpperCase();
  const date     = o.date ? new Date(o.date + 'T' + new Date().toTimeString().slice(0,8)) : new Date();

  };

/********************************
 * APPROVE / REJECT RETURN
 * Requires 'approve_returns'
 ********************************/
function approveReturn(o) {
  const sess = _verifySession(o._token, 'approve_returns');
  if(!sess) return _unauth();

  const ss           = SpreadsheetApp.getActive();
  const returnsSheet = ss.getSheetByName('Returns');
  if (!returnsSheet) return { success:false, msg:'Returns sheet not found' };

  const data = returnsSheet.getDataRange().getValues();
  let rowIndex = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === String(o.returnId).trim()) { rowIndex=i+1; break; }
  }
  if (rowIndex === -1) return { success:false, msg:'Return not found' };

  const now = new Date();

  if (o.action === 'approve') {
    returnsSheet.getRange(rowIndex, 13).setValue('Approved');
    returnsSheet.getRange(rowIndex, 14).setValue(o.approvedBy);
    returnsSheet.getRange(rowIndex, 15).setValue(now);

    if (!o.isWB) {
      const store = String(data[rowIndex-1][11]||'').trim();   // Store from the return row (#24)
      _restoreReturnStock(ss, store, o.productID, Number(o.qty));
    }

    const salesSheet = ss.getSheetByName('FactSales');
    if(salesSheet && o.originalTxnId){
      const salesData = salesSheet.getDataRange().getValues();
      for(let i=1;i<salesData.length;i++){
        const rowSaleId  = String(salesData[i][0]).trim();
        const rowProdId  = String(salesData[i][2]).trim();
        if(rowSaleId===String(o.originalTxnId).trim() && rowProdId===String(o.productID).trim()){
          const currentQty = Number(salesData[i][3]);
          const newQty     = currentQty - Number(o.qty);
          if(newQty<=0){ salesSheet.deleteRow(i+1); }
          else {
            const ratio     = newQty / currentQty;
            const origCost  = Number(salesData[i][5]) || 0;   // Total Cost  (col 6)
            const origTotal = Number(salesData[i][7]) || 0;   // TotalAmount (col 8, net of discount)
            salesSheet.getRange(i+1, 4).setValue(newQty);                        // Quantity
            salesSheet.getRange(i+1, 6).setValue(Math.round(origCost * ratio));  // Total Cost (#24)
            salesSheet.getRange(i+1, 8).setValue(Math.round(origTotal * ratio)); // TotalAmount, keeps discount ratio
          }
          break;
        }
      }
    }
    return { success:true, msg:'Return approved · Stock restored · Refund: ₦'+Number(o.refundAmount).toLocaleString() };
  } else {
    returnsSheet.getRange(rowIndex, 13).setValue('Rejected');
    returnsSheet.getRange(rowIndex, 14).setValue(o.approvedBy);
    returnsSheet.getRange(rowIndex, 15).setValue(now);
    return { success:true, msg:'Return rejected · '+o.returnId };
  }
}

      if(o.soldBy&&o.soldBy.trim()){ if(!soldBy.includes(o.soldBy.trim().toLowerCase())) continue; }
      if(o.store&&o.store.trim()){ if(store.toLowerCase()!==o.store.trim().toLowerCase()) continue; }
      results.push({ saleId, date:date.toISOString(), productId, productName, qty, totalAmount, paymentType:String(row[8]).trim(), soldBy:String(row[9]).trim(), store });
    }
    results.sort((a,b)=>new Date(b.date)-new Date(a.date));
    return results.slice(0,30);
  }catch(e){ return {error:e.message}; }
}

/********************************
 * ATTENDANCE OVERVIEW
 * Signature change: getAttendanceOverview(token)
 * Requires attendance_view OR staff_mgmt
 ********************************/
function getAttendanceOverview(token){
  const sess = _verifySession(token);
  if(!sess) return _unauth();
  if(!_hasAnyPerm(sess, ['attendance_view','staff_mgmt'])) return _unauth();

  const ss      = SpreadsheetApp.getActive();
  const attSh   = ss.getSheetByName('Attendance');
  const spSh    = ss.getSheetByName('SalesPerson');
  const shiftSh = ss.getSheetByName('Shifts');
  const today   = new Date();
  const todayStr= Utilities.formatDate(today, Session.getScriptTimeZone(), 'yyyy-MM-dd');

  const spData      = spSh.getDataRange().getValues();
  const spH         = headerMap(spData[0]);
  const activeStaff = spData.slice(1).filter(r=>r[spH.status]==='Active'&&r[spH.name]);
  const attData     = attSh.getDataRange().getValues();
  const attH        = headerMap(attData[0]);

  const todayAtt = attData.slice(1).filter(r=>{
    const d  = r[attH.date];
    const ds = d instanceof Date
      ? Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd')
      : String(d).split('T')[0];
    return ds===todayStr;
  });

  let shifts=[];
  if(shiftSh){
    const shData = shiftSh.getDataRange().getValues();
    const shH    = headerMap(shData[0]);
    shifts = shData.slice(1).map(r=>({
      staffId: String(r[shH.staffid]||r[shH.staff_id]||''),
      start:   r[shH.start]||r[shH.starttime]||'',
      grace:   parseInt(r[shH.grace]||r[shH.graceperiod]||5)
    }));
  }

  const records = activeStaff.map((s,i)=>{
    const name    = s[spH.name];
    const store   = s[spH.store];
    const role    = (s[spH.role]||'Staff').toLowerCase();
    const staffId = String(i+2);
    const att     = todayAtt.find(a=>String(a[attH.name])===String(name));
    const shift   = shifts.find(sh=>sh.staffId===staffId);
    let clockIn='', clockOut='', hours='', status='Absent';

    if(att){
      const cin  = att[attH.clockin]||att[attH['clock in']]||att[attH.timein]||'';
      const cout = att[attH.clockout]||att[attH['clock out']]||att[attH.timeout]||'';
      if(cin  instanceof Date) clockIn  = Utilities.formatDate(cin,  Session.getScriptTimeZone(), 'HH:mm');
      else clockIn  = String(cin).replace(/.*T/,'').slice(0,5)||String(cin);
      if(cout instanceof Date) clockOut = Utilities.formatDate(cout, Session.getScriptTimeZone(), 'HH:mm');
      else if(cout) clockOut = String(cout).replace(/.*T/,'').slice(0,5)||String(cout);
      if(clockIn&&clockOut){
        const [ih,im]=clockIn.split(':').map(Number);
        const [oh,om]=clockOut.split(':').map(Number);
        const mins=(oh*60+om)-(ih*60+im);
        if(mins>0) hours=Math.floor(mins/60)+'h '+(mins%60)+'m';
      }
      status='Present';
      if(shift&&shift.start&&clockIn){
        const startStr=shift.start instanceof Date
          ? Utilities.formatDate(shift.start, Session.getScriptTimeZone(), 'HH:mm')
          : String(shift.start);
        const [sh2,sm]=startStr.split(':').map(Number);
        const [ch,cm]=clockIn.split(':').map(Number);
        if(!isNaN(sh2)&&!isNaN(ch)&&(ch*60+cm)>(sh2*60+sm+(shift.grace||5))) status='Late';
      }
    }
    return {name, store, role, clockIn, clockOut, hours, status};
  });

  return {
    present: records.filter(r=>r.status==='Present').length,
    late:    records.filter(r=>r.status==='Late').length,
    absent:  records.filter(r=>r.status==='Absent').length,
    total:   activeStaff.length,
    records
  };
}

/********************************
 * ATTENDANCE HISTORY
 * Requires attendance_view OR staff_mgmt

function getAttendanceHistory(o){
  const sess = _verifySession(o._token);
  if(!sess) return _unauth();
  if(!_hasAnyPerm(sess, ['attendance_view','staff_mgmt'])) return _unauth();

  const ss      = SpreadsheetApp.getActive();
  const attSh   = ss.getSheetByName('Attendance');
  const spSh    = ss.getSheetByName('SalesPerson');
  const shiftSh = ss.getSheetByName('Shifts');
  const attData = attSh.getDataRange().getValues();
  const attH    = headerMap(attData[0]);
  const spData  = spSh.getDataRange().getValues();
  const spH     = headerMap(spData[0]);

  const staffMap={};
  spData.slice(1).forEach((r,i)=>{
    if(r[spH.name]) staffMap[String(r[spH.name])]={store:r[spH.store],role:r[spH.role],id:String(i+2)};
  });

  let shifts=[];
  if(shiftSh){
    const shData = shiftSh.getDataRange().getValues();
    const shH    = headerMap(shData[0]);
    shifts = shData.slice(1).map(r=>({
      staffId: String(r[shH.staffid]||r[shH.staff_id]||''),
      start:   r[shH.start]||r[shH.starttime]||'',
      grace:   parseInt(r[shH.grace]||r[shH.graceperiod]||5)
    }));
  }

  const from = new Date(o.from); from.setHours(0,0,0,0);
  const to   = new Date(o.to);   to.setHours(23,59,59,999);

  return attData.slice(1).filter(r=>{
    const d  = r[attH.date];
    const dt = d instanceof Date ? d : new Date(d);
    if(isNaN(dt)) return false;
    if(dt<from||dt>to) return false;
    if(o.staffId){
      const info = staffMap[String(r[attH.name])];
      if(!info||info.id!==o.staffId) return false;
    }
    return true;
  }).map(r=>{
    const name  = String(r[attH.name]);
    const info  = staffMap[name]||{store:'',role:''};
    const d     = r[attH.date];
    const date  = d instanceof Date
      ? Utilities.formatDate(d, Session.getScriptTimeZone(), 'dd MMM yyyy')
      : String(d).split('T')[0];
    const cin   = r[attH.clockin]||r[attH['clock in']]||r[attH.timein]||'';
    const cout  = r[attH.clockout]||r[attH['clock out']]||r[attH.timeout]||'';
    let clockIn='', clockOut='', hours='', status='Present';
    if(cin  instanceof Date) clockIn  = Utilities.formatDate(cin,  Session.getScriptTimeZone(), 'HH:mm');
    else clockIn = String(cin).replace(/.*T/,'').slice(0,5)||String(cin);
    if(cout instanceof Date) clockOut = Utilities.formatDate(cout, Session.getScriptTimeZone(), 'HH:mm');
    else if(cout) clockOut = String(cout).replace(/.*T/,'').slice(0,5);
    if(clockIn&&clockOut){
      const [ih,im]=clockIn.split(':').map(Number);
      const [oh,om]=clockOut.split(':').map(Number);
      const mins=(oh*60+om)-(ih*60+im);
      if(mins>0) hours=Math.floor(mins/60)+'h '+(mins%60)+'m';
    }
    const shift = shifts.find(sh=>sh.staffId===info.id);
    if(shift&&shift.start&&clockIn){
      const startStr = shift.start instanceof Date
        ? Utilities.formatDate(shift.start, Session.getScriptTimeZone(), 'HH:mm')
        : String(shift.start);
      const [sh2,sm]=startStr.split(':').map(Number);
      const [ch,cm]=clockIn.split(':').map(Number);
      if(!isNaN(sh2)&&!isNaN(ch)&&(ch*60+cm)>(sh2*60+sm+(shift.grace||5))) status='Late';
    }
    return {name, store:info.store, role:info.role, date, clockIn, clockOut, hours, status};
  });
}

function _upsertCustomer(name, wa, amount, product, store, saleID, date){
  const ss = SpreadsheetApp.getActive();
  let sh   = ss.getSheetByName('Customers');
  if(!sh){
    sh = ss.insertSheet('Customers');
    sh.appendRow(['Name','WhatsApp','TotalSpend','VisitCount','LastProduct','LastSaleID','LastVisit','FirstVisit','Store']);
  }
  const d      = sh.getDataRange().getValues();
  const h      = headerMap(d[0]);
  const txDate = date ? new Date(date + 'T12:00:00') : new Date();
  const dateStr= Utilities.formatDate(txDate, Session.getScriptTimeZone(), 'yyyy-MM-dd');

  for(let i=1;i<d.length;i++){
    if(
      String(d[i][h.name]).toLowerCase().trim()  === name.toLowerCase().trim() &&
      String(d[i][h.store]).toLowerCase().trim()  === store.toLowerCase().trim()
    ){
      sh.getRange(i+1, h.totalspend+1).setValue((Number(d[i][h.totalspend])||0)+amount);
      sh.getRange(i+1, h.visitcount+1).setValue((Number(d[i][h.visitcount])||0)+1);

    }
  }
  sh.appendRow([_sf(name), _sf(wa||''), amount, 1, _sf(product), saleID, dateStr, dateStr, _sf(store)]);
}

function getCustomerData(o){
  const sess = _verifySession(o._token, 'customers');
  if(!sess) return _unauth();

  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName('Customers');
  if(!sh) return { customers:[], stats:{total:0,returning:0,newThisMonth:0,activeThisWeek:0,totalSpend:0,avgSpend:0} };

  const d   = sh.getDataRange().getValues();
  const h   = headerMap(d[0]);
  const tz  = Session.getScriptTimeZone();
  const today = new Date();
  const filterStore = (o&&o.store) ? o.store.trim().toLowerCase().replace(/\s+/g,'') : '';
  const thisMonth   = Utilities.formatDate(today, tz, 'yyyy-MM');
  const sevenDaysAgo= new Date(today.getTime()-7*24*60*60*1000);

  const customers = d.slice(1).filter(r=>r[h.name]).map(r=>{
    const name  = String(r[h.name]||'').trim();
    const store = String(r[h.store]||'').trim();
    if(filterStore && store.toLowerCase().replace(/\s+/g,'')!==filterStore) return null;

    const totalSpend = Number(r[h.totalspend]||0);
    const visitCount = Number(r[h.visitcount]||0);

    const toDate = v=>{
      if(!v) return null;
      if(v instanceof Date) return isNaN(v)?null:v;
      const s = String(v).trim();
      if(/^\d{4}-\d{2}-\d{2}$/.test(s)){ const p=s.split('-'); return new Date(+p[0],+p[1]-1,+p[2]); }
      const d = new Date(s); return isNaN(d)?null:d;
    };

    const lastVisitDate  = toDate(r[h.lastvisit]);
    const firstVisitDate = toDate(r[h.firstvisit]);
    const daysSinceLastVisit = lastVisitDate ? Math.floor((today-lastVisitDate)/(1000*60*60*24)) : null;
    const lastVisitStr   = lastVisitDate  ? Utilities.formatDate(lastVisitDate,  tz, 'dd MMM yyyy') : '';
    const firstVisitStr  = firstVisitDate ? Utilities.formatDate(firstVisitDate, tz, 'dd MMM yyyy') : '';
    const firstVisitMonth= firstVisitDate ? Utilities.formatDate(firstVisitDate, tz, 'yyyy-MM') : '';

    return {
      name, whatsapp:String(r[h.whatsapp]||''), totalSpend, visitCount,
      avgSpendPerVisit: visitCount>0 ? Math.round(totalSpend/visitCount) : 0,
      favouriteProduct: String(r[h.lastproduct]||''),
      lastSaleID:       String(r[h.lastsaleid]||''),
      lastVisit:        lastVisitStr, firstVisit: firstVisitStr,
      daysSinceLastVisit, store,
      isReturning:     visitCount>=2,
      isNewThisMonth:  firstVisitMonth===thisMonth,
      isActiveThisWeek:lastVisitDate&&!isNaN(lastVisitDate)&&lastVisitDate>=sevenDaysAgo,
      allProducts:     r[h.lastproduct] ? [{name:String(r[h.lastproduct]),count:visitCount}] : []
    };
  }).filter(Boolean);

  const totalSpendAll = customers.reduce((s,c)=>s+c.totalSpend, 0);
  return {
    customers,
    stats:{
      total:          customers.length,
      returning:      customers.filter(c=>c.isReturning).length,
      newThisMonth:   customers.filter(c=>c.isNewThisMonth).length,
      activeThisWeek: customers.filter(c=>c.isActiveThisWeek).length,
      totalSpend:     totalSpendAll,
      avgSpend:       customers.length>0 ? Math.round(totalSpendAll/customers.length) : 0
    }
  };
}

function getExpiryAlerts(o){
  var sess=_verifySession(o._token);
  if(!sess) return _unauth();
  var ss=SpreadsheetApp.getActive();
  var stores=getAllStores();
  var today=new Date(); today.setHours(0,0,0,0);
  var alerts=[];
  stores.forEach(function(storeName){
    var sh=getStoreSheet(ss,storeName);
    if(!sh) return;
    var d=sh.getDataRange().getValues();
    var h=headerMap(d[0]);
    if(h.expirydate===undefined) return;
    d.slice(1).forEach(function(r){
      var stock=Number(r[h.stock])||0;
      if(stock<=0) return;
      var raw=r[h.expirydate];
      if(!raw||!r[h.productname]) return;
      var exp=new Date(raw); exp.setHours(0,0,0,0);
      var days=Math.round((exp-today)/86400000);
      if(days<=60){
        alerts.push({
          productID:r[h.productid],
          name:String(r[h.productname]),
          store:storeName,
          stock:stock,
          expiryDate:Utilities.formatDate(exp,Session.getScriptTimeZone(),'yyyy-MM-dd'),
          daysToExpiry:days,
          status:days<0?'expired':days<=7?'critical':days<=14?'warning':'soon'
        });
      }
    });
  });
  return alerts.sort(function(a,b){return a.daysToExpiry-b.daysToExpiry;});
}

function checkExpiryAlerts(){
  var ss=SpreadsheetApp.getActive();
  var stores=getAllStores();
  var today=new Date(); today.setHours(0,0,0,0);
  var sentSh=_getSentAlertsSheet(ss);
  var sentData=sentSh.getDataRange().getValues();
  var sH=headerMap(sentData[0]);
  var newAlerts=[];
  stores.forEach(function(storeName){
    var sh=getStoreSheet(ss,storeName);
    if(!sh) return;
    var d=sh.getDataRange().getValues();
    var h=headerMap(d[0]);
    if(h.expirydate===undefined) return;
    d.slice(1).forEach(function(r){
      var stock=Number(r[h.stock])||0;
      if(stock<=0) return;
      var raw=r[h.expirydate];
      if(!raw||!r[h.productname]) return;
      var exp=new Date(raw); exp.setHours(0,0,0,0);
      var days=Math.round((exp-today)/86400000);
      var pid=String(r[h.productid]);
      var pname=String(r[h.productname]);
      var expStr=Utilities.formatDate(exp,Session.getScriptTimeZone(),'yyyy-MM-dd');
      var rawThresh=r[h.expiryalertdays]?String(r[h.expiryalertdays]):'14';
      var thresholds=rawThresh.split(',').map(function(x){return parseInt(x.trim());}).filter(Boolean);
      thresholds.forEach(function(thresh){
        if(days<=thresh&&days>=0){
          var sent=sentData.slice(1).some(function(row){
            return String(row[sH.productid])===pid&&
                   String(row[sH.store])===storeName&&
                   String(row[sH.expirydate]).slice(0,10)===expStr&&
                   Number(row[sH.alertdays])===thresh;
          });
          if(!sent){
            newAlerts.push({
              name:pname, store:storeName, expiryDate:expStr,
              daysToExpiry:days, thresh:thresh,
              _row:[pid,pname,storeName,expStr,thresh,
                Utilities.formatDate(today,Session.getScriptTimeZone(),'yyyy-MM-dd'),false]
            });
          }
        }
      });
      if(days<0){
        var sentExp=sentData.slice(1).some(function(row){
          return String(row[sH.productid])===pid&&
                 String(row[sH.store])===storeName&&
                 String(row[sH.expirydate]).slice(0,10)===expStr&&
                 Number(row[sH.alertdays])===-1;
        });
        if(!sentExp){
          newAlerts.push({
            name:pname, store:storeName, expiryDate:expStr,
            daysToExpiry:days, thresh:-1,
            _row:[pid,pname,storeName,expStr,-1,
              Utilities.formatDate(today,Session.getScriptTimeZone(),'yyyy-MM-dd'),false]
          });
        }
      }
    });
  });
  if(newAlerts.length){
    var sent=_sendExpiryEmail(newAlerts);
    if(sent) newAlerts.forEach(function(a){ sentSh.appendRow(a._row); });
  }
  return {checked:true,alerts:newAlerts.length};
}

    html+='<td style="padding-left:8px;font-size:16px;font-weight:800;color:#dc2626;vertical-align:middle">Expired — Still in Stock</td>';

    html+='<td style="padding-left:8px;vertical-align:middle"><span style="background:#fef3c7;color:#d97706;font-size:11px;font-weight:800;padding:2px 8px;border-radius:20px">'+expiring.length+'</span></td>';
    html+='</tr></table>';
    html+='<p style="color:#6b7280;font-size:13px;margin:6px 0 12px">These products are approaching their expiry date.</p>';
    html+='</td></tr>';
    expiring.forEach(function(a){ html+=productRow(a,false); });
    html+='<tr><td style="padding-bottom:20px"></td></tr>';
  }

  // What to do
  html+='<tr><td>';
  html+='<table width="100%" cellpadding="0" cellspacing="0" style="background:#eff6ff;border:1px solid #bfdbfe;border-radius:8px">';
  html+='<tr><td style="padding:18px 20px">';
  html+='<p style="font-size:14px;font-weight:800;color:#1e40af;margin:0 0 10px">📋 Recommended Actions</p>';
  if(expired.length){
    html+='<p style="font-size:13px;color:#374151;margin:0 0 6px"><strong style="color:#dc2626">For expired products:</strong></p>';
    html+='<ul style="margin:0 0 10px;padding-left:18px;color:#6b7280;font-size:13px">';
    html+='<li style="margin-bottom:4px">Check if the product is still on the shelf</li>';
    html+='<li style="margin-bottom:4px">Remove expired stock from display immediately</li>';
    html+='<li style="margin-bottom:4px">Record disposal or write-off in the system</li>';
    html+='</ul>';
  }
  if(expiring.length){

    html+='<ul style="margin:0;padding-left:18px;color:#6b7280;font-size:13px">';

    html+='<li style="margin-bottom:4px">Do not restock until current batch is cleared</li>';

  }

  html+='</table>';
  html+='</td></tr>';

  // Footer
  html+='<tr><td style="background:#f8fafd;border:1px solid #dde3f0;border-top:none;border-radius:0 0 12px 12px;padding:16px 24px;text-align:center">';
  html+='<p style="font-size:12px;color:#9ca3af;margin:0">Automated alert from <strong>StarTrack POS</strong></p>';
  html+='<p style="font-size:11px;color:#d1d5db;margin:4px 0 0">Please do not reply to this email</p>';
  html+='</td></tr>';

  html+='</table></td></tr></table></body></html>';

  var ok=true;
  emails.forEach(function(email){
    try{ _sendEmail(email,'Expiry Alert — '+(expired.length?expired.length+' Expired, ':'')+expiring.length+' Expiring Soon',html); }
    catch(e){ ok=false; Logger.log('Expiry email failed: '+e.message); }
  });
  return ok;
}

function clearProductExpiry(o){
  var sess=_verifySession(o._token);
  if(!sess) return _unauth();
  var ss=SpreadsheetApp.getActive();
  var sh=getStoreSheet(ss,o.store);
  if(!sh) return {success:false,msg:'Store not found'};
  var d=sh.getDataRange().getValues();
  var h=headerMap(d[0]);
  for(var i=1;i<d.length;i++){
    if(String(d[i][h.productid])===String(o.productID)){
      if(h.expirydate!==undefined) sh.getRange(i+1,h.expirydate+1).setValue('');
      break;
    }
  }
  _autoDismissExpiryAlerts(ss,o.productID,o.store);
  return {success:true};
}
